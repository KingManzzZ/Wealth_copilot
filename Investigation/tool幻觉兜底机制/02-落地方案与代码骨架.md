_READ_ONLY_PREFIXES = ("get_", "calculate_", "search_", "list_")

#: 自动修复的相似度门槛（difflib ratio）。高于"建议"档，因为这里**会直接执行**。
_AUTO_REPAIR_CUTOFF = 0.85

#: 仅用于候选建议的门槛，只要求"看起来像"。
_SUGGEST_CUTOFF = 0.5

#: 候选建议最多返回几条
_SUGGEST_LIMIT = 3

### 1.2 名称归一化

```python
_CAMEL_RE = re.compile(r"(?<=[a-z0-9])(?=[A-Z])")
_SEP_RE = re.compile(r"[\s\-./:]+")
_MULTI_UNDERSCORE_RE = re.compile(r"_+")
_NAMESPACE_RE = re.compile(r"^(?:functions?|tools?)\s*[.:]\s*", re.IGNORECASE)


def normalize_name(name: str) -> str:
    """把模型给出的工具名归一到注册表使用的 snake_case 形式。

    处理四类常见书写漂移：

    - 命名空间前缀：``functions.get_assets`` → ``get_assets``
    - 驼峰：        ``getAssets``            → ``get_assets``
    - 大小写：      ``Get_Assets``           → ``get_assets``
    - 分隔符：      ``get-assets`` / ``get assets`` → ``get_assets``
    """
    s = str(name or "").strip()
    s = _NAMESPACE_RE.sub("", s)
    s = _CAMEL_RE.sub("_", s)
    s = s.lower()
    s = _SEP_RE.sub("_", s)
    return _MULTI_UNDERSCORE_RE.sub("_", s).strip("_")
```

### 1.3 解析阶梯与"写工具不自动修复"

```python
#: 归一化索引：normalize_name(工具名) → 注册表里的真名
_NORMALIZED: dict = {}


def _rebuild_normalized_index() -> None:
    _NORMALIZED.clear()
    for name in TOOL_REGISTRY:
        _NORMALIZED.setdefault(normalize_name(name), name)


def _is_read_only(tool_name: str) -> bool:
    return tool_name.startswith(_READ_ONLY_PREFIXES)


def resolve_tool_name(tool_name: str) -> tuple[str | None, str]:
    """把模型给的名字解析成注册表里的真名。

    返回 ``(真名 或 None, 命中方式)``，命中方式取值：
    ``exact`` / ``normalized`` / ``fuzzy`` / ``none``。

    **模糊匹配只对只读工具开放**：把 ``update_asset`` 猜成 ``update_debt``
    会静默执行一个错误的写操作——调用成功、没有报错、数据已改坏。
    代价远大于"多问模型一轮"，所以写工具只给建议、不自动修复。
    """
    if tool_name in TOOL_REGISTRY:
        return tool_name, "exact"

    normalized = normalize_name(tool_name)
    hit = _NORMALIZED.get(normalized)
    if hit is not None:
        return hit, "normalized"

    matches = difflib.get_close_matches(
        normalized, _NORMALIZED.keys(), n=1, cutoff=_AUTO_REPAIR_CUTOFF
    )
    if matches:
        resolved = _NORMALIZED[matches[0]]
        if _is_read_only(resolved):
            return resolved, "fuzzy"
    return None, "none"


def _suggest_tool_names(tool_name: str) -> list[str]:
    """给出候选工具名（只做建议，绝不据此执行）。"""
    matches = difflib.get_close_matches(
        normalize_name(tool_name), _NORMALIZED.keys(), n=_SUGGEST_LIMIT, cutoff=_SUGGEST_CUTOFF
    )
    return [_NORMALIZED[m] for m in matches]
```

### 1.4 一个实测发现：候选建议无法靠阈值区分好坏 ⚠️

用本项目真实的 24 个工具名跑过之后，发现一件值得记下来的事。下面两组相似度是**反直觉的**：

| 输入 → 候选 | ratio |
|---|---|
| `search_policy` → `search_knowledge`（**合理的近失**） | 0.621 |
| `nope_not_a_tool` → `complete_goal_tool`（**纯噪声**） | 0.667 |

噪声的相似度**比合理近失还高**。原因很直接：工具名共享领域词汇（`get_`、`_tool`、`_asset`、`_debt`），`nope_not_a_tool` 与 `complete_goal_tool` 共享了 `_tool` 后缀和若干字母，difflib 就给了 0.667。

**结论**：不要试图用"提高阈值"来去掉噪声建议——那会先杀掉合理的那条。正确的处理方式是：

- 候选建议**只作为提示**，交给模型判断，绝不据此执行（这正是 1.3 里 `_suggest_tool_names` 与 `resolve_tool_name` 分成两个函数的原因）；
- 接受"有时会给出噪声建议"这个代价。模型看到 `nope_not_a_tool` 的候选是 `complete_goal_tool` 时不会去调它，而看到 `search_policy` 的候选是 `search_knowledge` 时会立刻改对——**收益是不对称的**。

这条也解释了为什么 4.1 节的反向用例（不自动修复写工具）是整个方案的安全底线：一旦让候选建议参与执行，噪声就会变成真实的错误调用。

### 1.5 参数校验（用 schema，不用 Python 报错）

```python
#: 工具名 → 该工具的 parameters 声明
_TOOL_SPECS: dict = {
    item["function"]["name"]: item["function"].get("parameters") or {}
    for item in tools_schema.TOOLS_SCHEMA
    if item.get("type") == "function"
}


def validate_arguments(tool_name: str, kwargs) -> dict | None:
    """按 schema 校验参数；有问题返回 error 载荷，没问题返回 None。

    校验三件事：是不是对象、必填项齐不齐、有没有幻觉参数。
    **不做类型校验**——成本高、收益低，多数工具自身能容忍。
    """
    if not isinstance(kwargs, dict):
        return {
            "error": f"参数必须是 JSON 对象，收到 {type(kwargs).__name__}",
            "retryable": True,
            "hint": '请输出形如 {"key": value} 的 JSON 对象。',
        }

    params = _TOOL_SPECS.get(tool_name) or {}
    required = list(params.get("required") or [])
    properties = params.get("properties") or {}

    missing = [name for name in required if name not in kwargs]
    if missing:
        return {
            "error": f"缺少必填参数: {', '.join(missing)}",
            "missing_arguments": missing,
            "required_arguments": required,
            "retryable": True,
        }

    unknown = [name for name in kwargs if name not in properties]
    if unknown:
        return {
            "error": f"不支持的参数: {', '.join(unknown)}",
            "unknown_arguments": unknown,
            "allowed_arguments": list(properties),
            "retryable": True,
        }

    return None
```

### 1.6 新的 `route()`

在模块末尾、`TOOL_REGISTRY, _UNDECLARED = _build_registry()` **之后**插入 1.2–1.4 的代码，并**在其后**调用一次 `_rebuild_normalized_index()`，然后替换 `route()`：

```python
_rebuild_normalized_index()


def route(tool_name: str, arguments: str) -> str:
    """路由：解析名字 → 校验参数 → 执行 → 返回 JSON 字符串。

    任何失败都返回 ``{"error": ...}`` 而**不抛异常**：Agent 循环依赖这一点，
    才能在工具出错后继续与模型对话（见 agent.py 的 tool 消息回填）。

    错误载荷刻意带上 ``available_tools`` 与 ``did_you_mean``：
    模型看到清单后重新选择，比在没有新信息的情况下重采样有效得多
    （这正是 hermes-agent 把"盲重试 3 次"改成"喂错误信息"的原因）。
    """
    resolved, how = resolve_tool_name(tool_name)

    if resolved is None:
        payload = {
            "error": f"未知工具: {tool_name}",
            "unknown_tool": True,
            "retryable": True,
            "did_you_mean": _suggest_tool_names(tool_name),
            "available_tools": sorted(TOOL_REGISTRY),
            "hint": "请从 available_tools 中选择工具名后重试，不要臆造工具名。",
        }
        print(f"[tools_route] ⚠️ 未知工具 tool={tool_name!r}")
        return json.dumps(payload, ensure_ascii=False)

    if how != "exact":
        # 修好了也要记：这是"本不该失败"的信号，值得单独统计
        print(f"[tools_route] 🔧 工具名已修复 tool={tool_name!r} -> {resolved!r} how={how}")

    try:
        kwargs = json.loads(arguments) if arguments else {}
    except json.JSONDecodeError as exc:
        return json.dumps(
            {
                "error": f"参数不是合法 JSON：{exc.msg}",
                "error_kind": "bad_json",
                "retryable": True,
                "hint": "请输出一个 JSON 对象作为参数。",
            },
            ensure_ascii=False,
        )

    problem = validate_arguments(resolved, kwargs)
    if problem is not None:
        problem["error_kind"] = "bad_arguments"
        return json.dumps(problem, ensure_ascii=False)

    try:
        return json.dumps(TOOL_REGISTRY[resolved](**kwargs), ensure_ascii=False)
    except Exception as exc:  # noqa: BLE001
        # 工具内部异常默认视为**不可重试**（大概率是确定性 bug 或依赖缺失）。
        # 真正的临时故障需要工具侧抛专门异常类型才能区分，见调研文档 5.3。
        print(f"[tools_route] ❌ 工具执行失败 tool={resolved!r} exc={type(exc).__name__}: {exc}")
        return json.dumps(
            {
                "error": f"{type(exc).__name__}: {exc}",
                "error_kind": "tool_exception",
                "retryable": False,
                "hint": "该错误大概率无法通过重试解决，请改用其他方式回答或直接告知用户。",
            },
            ensure_ascii=False,
        )
```

**行为变化对照**

| 输入 | 改前 | 改后 |
|---|---|---|
| `Search_Knowledge` + `{"query": "x"}` | 未知工具 | 归一化命中，正常执行 |
| `get-assets` | 未知工具 | 归一化命中 |
| `get_assets `（尾空格） | 未知工具 | 归一化命中 |
| `get_asset`（少个 s） | 未知工具 | 模糊命中（只读），执行 `get_assets` |
| `update_asset` → 拼错 | 未知工具 | **不自动修复**，返回 `did_you_mean` 让模型决定 |
| `get_assets` + `{}` | 正常执行 | 正常执行（无必填项） |
| `search_knowledge` + `{}` | `TypeError` 原文 | `缺少必填参数: query` |
| `get_assets` + `{"foo": 1}` | `TypeError` 原文 | `不支持的参数: foo` |
| `get_assets` + `[1,2]` | `argument of type ... is not a mapping` | `参数必须是 JSON 对象，收到 list` |

## 3. Phase 3：`prompt.py` 增补

在 `## 关于拒绝` 之前插入：

```text
## 关于工具调用

* 只能使用系统提供的工具。工具名必须与提供的清单完全一致，
  不要臆造、缩写、翻译或改写工具名。
* 调用工具前先确认参数齐全。若缺少必需信息（如金额、日期、标的名称），
  应当先向用户询问，而不是自行假设一个值填入。
* 如果现有工具都不足以完成当前任务，直接用文字向用户说明情况，
  不要为了"显得在工作"而调用不相关的工具，更不要编造工具名。
* 同一个工具在同样的参数下不要重复调用；如果需要更多信息，
  应当调整参数或换用其他工具。
```

最后一条对应"时机幻觉"，前两条对应"类型幻觉/内容幻觉"，第三条是**弃权许可**——调研里 Relign 发现，模型的默认倾向是"必须调点什么"，明确给它一个"什么都不调、直接说明"的出口能显著降低硬协概率。

def check_route_rejects_non_object_arguments():

def _two_unknown_rounds():
    """两轮都调用不存在的工具 → 应在第 2 轮后收敛，不跑满 MAX_ROUNDS。"""

def check_unknown_tool_breaker_stops_early():

    mod = _load_agent(fake)
    events = _collect(mod.chat_stream("查点东西"))

    eq(len(fake.requests), 2, "连续 2 次未知工具后应停止，不应再请求模型")
    errors = [e for e in events if e["type"] == "error"]
    true(any("不存在的工具" in e["message"] for e in errors),
         "熔断文案应归因到'工具名不存在'")
    eq(events[-1]["type"], "done", "熔断后仍应以 done 收尾")


def check_duplicate_tool_call_is_skipped():
    """同名同参第二次出现应被跳过，且不真正执行。"""
    from core.tools import tools_route

    calls = []
    real_route = tools_route.route

    def fake_route(name, args):
        calls.append((name, args))
        return json.dumps({"ok": True})

    tools_route.route = fake_route
    try:
        mk = lambda cid: [
            chunk(_Delta(content="查一下。")),
            chunk(_Delta(tool_calls=[_tool_call_delta(0, cid, "get_assets", "{}")])),
        ]
        fake = FakeOpenAI([mk("call_1"), mk("call_2"), *_answer_only("完成")])
        mod = _load_agent(fake)
        events = _collect(mod.chat_stream("q"))
    finally:
        tools_route.route = real_route

    eq(len(calls), 1, "同一个 (name, args) 只应真正执行一次")
    results = [e for e in events if e["type"] == "tool_result"]
    eq(len(results), 2, "两次调用都应产生 tool_result（第二次是跳过提示）")
    eq(results[-1]["result"]["error_kind"], "duplicate_call")


def check_non_dict_arguments_downgraded():

## 6. 埋点与后续演进

### 6.1 已有埋点（P0/P1 自带）

```
[tools_route] ⚠️ 未知工具 tool='search_policy'
[tools_route] 🔧 工具名已修复 tool='Get_Assets' -> 'get_assets' how=normalized
[tools_route] ❌ 工具执行失败 tool='search_knowledge' exc=RuntimeError: ...
```

建议统一成一条结构化日志，便于统计：

```text
[tools_route] tool=<原名> resolved=<真名> how=<exact|normalized|fuzzy|none>
              status=<success|error> error_kind=<unknown_tool|bad_json|bad_arguments|
              tool_exception|duplicate_call>
```

### 6.2 三个需要持续看的数

| 指标 | 怎么看 | 低了怎么办 |
|---|---|---|
| 未知工具率 | `unknown_tool` 次数 / 总调用 | 检查 prompt 是否列了清单 |
| **自愈率** | 首次失败后下一轮成功 / 首次失败次数 | 低了说明错误信息不够（回到 `did_you_mean`） |
| `how=normalized` 占比 | 修复日志 | 占比高说明模型对写法不敏感，可进一步放宽 |

### 6.3 演进方向

- **别名表**：把线上高频错名沉淀成显式映射（`search_policy` → `search_knowledge`）。前提是先有 6.1 的数据，别拍脑袋。
- **工具级异常分级**：让工具抛 `RetryableToolError` / `FatalToolError`，`route()` 据此填 `retryable`，替代现在"工具异常一律不可重试"的粗判。
- **工具数 > 50**：引入 Tool RAG（朴素向量召回起步；有依赖关系再上 Graph RAG-Tool Fusion 的思路）。
- **自建推理后**：启用约束解码（vLLM / TensorRT-LLM + XGrammar），把工具名做成语法枚举——这是唯一能让名字幻觉**归零**的手段。
